CREATE DATABASE StudentDB;
-- Step 2: Use the created database
USE StudentDB;
-- Step 3: Create a table
CREATE TABLE Students (
student_id INT AUTO_INCREMENT PRIMARY KEY,
name VARCHAR(50),
age INT,
grade VARCHAR(10)
);
