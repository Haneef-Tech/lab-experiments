#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <sys/wait.h>

int main()
{
    char command[100];

    while(1)
    {
        printf("MyShell$ ");
        fgets(command, sizeof(command), stdin);

        command[strcspn(command, "\n")] = '\0';

        if(strcmp(command, "exit") == 0)
            break;

        if(fork() == 0)
        {
            execlp(command, command, NULL);
            printf("Command not found!\n");
            exit(1);
        }
        else
        {
            wait(NULL);
        }
    }

    return 0;
}
